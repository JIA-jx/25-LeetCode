## [236. 二叉树的最近公共祖先 - 力扣（LeetCode）](https://leetcode.cn/problems/lowest-common-ancestor-of-a-binary-tree/)

![屏幕截图 2026-04-02 202754.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-04-02%20202754.png?raw=true)

```cpp
class Solution {
public:
    TreeNode* result; // 存储最近公共祖先

    // DFS：判断子树是否包含p/q，同时定位LCA
    bool dfs(TreeNode* root, TreeNode* p, TreeNode* q){
        if(!root) return false;

        bool l = dfs(root->left, p, q);  // 左子树是否含p/q
        bool r = dfs(root->right, p, q); // 右子树是否含p/q
        
        // 判定为LCA：左右各含一个 / 自身是p/q且子树含另一个
        if(l && r || ((l || r) && (root == p || root == q))){
            result = root;
        }
        
        // 返回：当前子树是否包含p/q
        return l || r || (root == p || root == q);
    }

    // 二叉树最近公共祖先
    TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
        dfs(root, p, q);
        return result;
    }
};
```



## [501. 二叉搜索树中的众数 - 力扣（LeetCode）](https://leetcode.cn/problems/find-mode-in-binary-search-tree/)

![屏幕截图 2026-04-02 202741.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-04-02%20202741.png?raw=true)

```cpp
class Solution {
public:
    vector<int> ans; // 存储中序遍历结果（BST有序）

    // 寻找二叉搜索树的众数
    vector<int> findMode(TreeNode* root) {
        vector<int> result;
        if(!root) return result;
        
        inorder(root); // 中序遍历得到有序序列

        // 遍历有序序列，统计频次找众数
        int pre = ans.front();
        int count = 0, max = INT_MIN;
        int n = ans.size();
        for(int i = 0; i < n; i++){
            if(pre == ans[i]) count++;
            else{
                count = 1;
                pre = ans[i];
            }
            // 更新最大频次与结果
            if(max < count){
                max = count;
                result.clear();
                result.push_back(pre);
            }
            else if(max == count) result.push_back(pre);
        }
        return result;
    }

    // BST中序遍历：左-根-右
    void inorder(TreeNode* cur){
        if(!cur) return;
        inorder(cur->left);
        ans.push_back(cur->val);
        inorder(cur->right);
    }
};
```
