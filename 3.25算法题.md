## [148. 排序链表 - 力扣（LeetCode）](https://leetcode.cn/problems/sort-list/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.25-1-2026-03-25-21%3A44%3A25.png)

```cpp
class Solution {
public:
    ListNode* sortList(ListNode* head) {//归并排序思路
        if(head == nullptr || head->next == nullptr) return head;
        ListNode* head2 = middleNode(head);//将链表从中间分开
        head = sortList(head);//前半部分归并
        head2 = sortList(head2);//后半部分归并
        return merge(head,head2);//合并两个链表
    }
private:
    ListNode* middleNode(ListNode* head){
        ListNode* pre = head;//拆分链表的前一个节点
        ListNode* slow = head;
        ListNode* fast = head;
        while(fast && fast->next){//快指针走到尾，慢指针走到中间
            pre = slow;
            slow = slow->next;
            fast = fast->next->next;
        }
        pre->next = nullptr;//在中间断开
        return slow;//后半部分的头节点
    }
    ListNode* merge(ListNode* list1,ListNode* list2){
        ListNode dummy;//虚拟头节点
        ListNode* cur = &dummy;
        while(list1 && list2){//两个都不为空
            if(list1->val < list2->val){//将小的值取出来
                cur->next = list1;
                list1 = list1->next;
            }else{
                cur->next = list2;
                list2 = list2->next;
            }
            cur = cur->next;
        }
        cur->next = list1 ? list1 : list2;//若有一个链表没有节点，则将有节点的链表拼在后面
        return dummy.next;
    } 
};
```

## [102. 二叉树的层序遍历 - 力扣（LeetCode）](https://leetcode.cn/problems/binary-tree-level-order-traversal/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.25-2-2026-03-25-22%3A05%3A44.png)

```cpp
class Solution {
public:
    vector<vector<int>> levelOrder(TreeNode* root) {
        vector<vector<int>> res;//记录结果
        if(!root){
            return res;
        }
        queue<TreeNode*> q;//队列处理临时存放
        q.push(root);//根节点入队
        while(!q.empty()){
            int cur = q.size();//当前层节点数
            res.push_back(vector<int>());//存储节点值
            for(int i = 1; i <= cur;++i){//加入节点值进入蹭列表
                auto node = q.front();q.pop();
                res.back().push_back(node->val);
                if(node -> left) q.push(node->left);//节点存在入队
                if(node->right) q.push(node->right);
            }
        }
        return res;
    }
};
```
