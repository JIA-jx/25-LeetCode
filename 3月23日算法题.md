## [647. 回文子串 - 力扣（LeetCode）](https://leetcode.cn/problems/palindromic-substrings/description/)

![%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-03-23%20195155.png](https://raw.githubusercontent.com/zhanglihang666/leetcodeImage/fae3eba02108577972c4044b05edc0088e523c1c/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-03-23%20195155.png)

```cpp
class Solution {
public:
    int countSubstrings(string s) {
        string t = "#";
        //用#分割s的每一个字符
        for(int i = 0; i < s.size(); i++){
            t += s[i];
            t += '#';
        }
        int n = t.size();
        vector<int> p(n, 0);
        int c = 0;//最右边回文串的中心点
        int R = 0;//最右边回文串的右边界
        int count = 0;//回文串数量
        for(int i = 0; i < n; i++){
            //如果当前字符属于最长回文串的一部分，就可以直接用镜像点已经算过的部分
            if(i < R){
                p[i] = min(p[2*c - i], R-i);
            }
            //继续扩展延长回文串
            while(i + p[i] + 1 < n && i - p[i] - 1 >= 0 && t[i + p[i] + 1] == t[i - p[i] - 1]){
                p[i]++;
            }
			//一旦超过最右边回文串的右边界，就更新右边界和中心点
            if(i + p[i] > R){
                R = i + p[i];
                c = i;
            }
            //累加回文串数量
            count += (p[i]+1)/2;   
        }
        return count;
    }
};
```





## [516. 最长回文子序列 - 力扣（LeetCode）](https://leetcode.cn/problems/longest-palindromic-subsequence/description/)

![%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-03-23%20195217.png](https://raw.githubusercontent.com/zhanglihang666/leetcodeImage/fae3eba02108577972c4044b05edc0088e523c1c/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-03-23%20195217.png)

```cpp
class Solution {
public:
    int longestPalindromeSubseq(string s) {
        //dp[i][j]表示i到j的回文串长度
        vector<vector<int>> dp(s.size(), vector<int>(s.size(), 0));
        for(int i = s.size() - 1; i >= 0; i--){
            for(int j = i; j < s.size(); j++){
                //两边扩展字符相等就取扩展前的回文串长度加2
                if(s[i] == s[j] && i != j){
                    dp[i][j] = dp[i+1][j-1] + 2;
                }
                //一个字符必是回文串
                else if(i == j){
                    dp[i][j] = 1;
                }
                //取任意删除两边一个字符的最长回文串长度
                else{
                    dp[i][j] = max(dp[i+1][j], dp[i][j-1]);
                }
            }
        }

        return dp[0][s.size()-1];
    }
};
```
