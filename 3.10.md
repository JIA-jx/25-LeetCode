## [142. 环形链表 II ](https://leetcode.cn/problems/linked-list-cycle-ii/)

struct ListNode *detectCycle(struct ListNode *head) {
    struct ListNode* fast = head;  //快慢指针
    struct ListNode* slow = head;
    while (fast && fast->next) {  //慢指针走一步，快指针走两步
        slow = slow->next;
        fast = fast->next->next;
        if (slow == fast) {  //相遇
            struct ListNode* ptr = head;  //在头节点设置一个指针ptr
            while (ptr != slow) {  //ptr和慢指针同时走，两者走到相同时，该节点恰为入环口
                ptr = ptr->next;
                slow = slow->next;
            }
            return ptr;  //返回入环口
        }
    }
    return NULL;
}

## [20. 有效的括号 ](https://leetcode.cn/problems/valid-parentheses/submissions/705352652/)

![image-20260310220407675](C:\Users\OMEN\AppData\Roaming\Typora\typora-user-images\image-20260310220407675.png)

```c
char pairs(char a) {
    if(a == '}') return '{';
    if(a == ']') return '[';
    if(a == ')') return '(';
    return 0;
}

bool isValid(char* s) {
    int n = strlen(s);
    if(n % 2 == 1) {  //长度为奇数时一定不闭合
        return false;
    }

    int stack[n + 1], top = 0;
    for(int i = 0; i < strlen(s); i++) {
        char ch = pairs(s[i]);
        if(ch) {
            if(top == 0 || stack[--top] != ch) {  //判断栈中还有没有左括号，左括号是否与当前匹配
                return false;
            }
        } else{
            stack[top++] = s[i];
        }
    }
    return top == 0;  //判断栈中是否还有剩余的左括号
}
```
