## [46. 全排列 - 力扣（LeetCode）](https://leetcode.cn/problems/permutations/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.31-1-2026-03-31-21%3A22%3A37.png)

```cpp
class Solution {
public:
    vector<vector<int>> permute(vector<int>& nums) {
        dfs(nums,0);//进行递归
        return res;
    }
private:
    vector<vector<int>> res;//结果数组
    void dfs(vector<int> nums, int x){
        if(x == nums.size() - 1){//处理完所有数字后终止递归
            res.push_back(nums);//压入结果
            return;
        }
        for(int i = x; i < nums.size(); i++){
            swap(nums[i], nums[x]);//交换、固定当前位置
            dfs(nums, x + 1);//开始固定下一个位置，深度递归
        }
    }
};
```

## [78. 子集 - 力扣（LeetCode）](https://leetcode.cn/problems/subsets/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.31-2-2026-03-31-21%3A43%3A50.png)

```cpp
class Solution {
public:
    vector<vector<int>> subsets(vector<int>& nums) {
        int n = nums.size();//记录数组大小
        vector<vector<int>> res;//记录结果
        vector<int> path;//记录每一个子集
        auto dfs = [&](this auto && dfs, int i)->void{
            if(i == n){//当所有数字被遍历，终止递归
                res.emplace_back(path);//将小子集放入结果
                return;
            }
            dfs(i + 1);//更深一步递归（不添加该元素）
            path.push_back(nums[i]);//向子串中加入上一个值
            dfs(i + 1);//选择第i个值（添加该元素）
            path.pop_back();//撤销上次添加结果
        };
        dfs(0);//从0开始进行递归
        return res;
    }
};
```
