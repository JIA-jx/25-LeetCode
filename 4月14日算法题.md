## [5. 最长回文子串 - 力扣（LeetCode）](https://leetcode.cn/problems/longest-palindromic-substring/?envType=problem-list-v2&envId=string)

![屏幕截图 2026-04-14 204238.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-04-14%20204238.png?raw=true)

```cpp
class Solution {
public:
    // Manacher算法：求最长回文子串
    string longestPalindrome(string s) {
        // 预处理：插入#，统一奇偶回文
        string t = "#";
        for(int i = 0; i < s.size(); i++){
            t += s[i];
            t += "#";
        }

        vector<int> p(t.size(), 0); // p[i]：i为中心的最长回文半径
        int max_c = 0, max_r = 0;   // 最长回文的中心、半径
        int C = 0, R = 0;           // 当前回文中心、右边界

        for(int i = 0; i < t.size(); i++){
            int mirror = 2*C - i;   // i关于C的镜像点
            if(i < R) p[i] = min(R - i, p[mirror]); // 利用镜像优化

            // 中心扩展
            int j = i+p[i]+1, k = i-p[i]-1;
            while(j < t.size() && k >= 0 && t[j] == t[k]){
                j++, k--;
            }
            p[i] = j-i-1;

            // 更新回文中心和右边界
            if(i + p[i] > R){
                R = i+p[i];
                C = i;
            }

            // 记录最长回文
            if(p[i] > max_r){
                max_r = p[i];
                max_c = i;
            }
        }

        // 截取原串结果
        return s.substr((max_c-max_r)/2, max_r);
    }
};
```

## [6. Z 字形变换 - 力扣（LeetCode）](https://leetcode.cn/problems/zigzag-conversion/submissions/718189100/?envType=problem-list-v2&envId=string)

![屏幕截图 2026-04-14 212807.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-04-14%20212807.png?raw=true)

```cpp
class Solution {
public:
    string convert(string s, int numRows) {
        // 边界：1行/行数超长度，直接返回原串
        if(numRows == 1 || numRows >= s.size()){
            return s;
        }
        // 二维数组暂存Z字形字符
        vector<vector<char>> temp(numRows, vector<char>(1001, '0'));
        
        // i遍历字符串，j行标，k列标
        for(int i = 0, j = 0, k = 0; i < s.size(); i++){
            temp[j][k] = s[i];
            // 到达底部，向上填充
            if(j == numRows-1){
                j--;
                k++;
            }
            // 到达顶部，向下填充
            else if(j == 0){
                j++;
            }
            // 中间向上填充
            else if(temp[j-1][k] == '0'){
                j--;
                k++;
            }
            // 中间向下填充
            else{
                j++;
            }
        }

        // 遍历二维数组，拼接结果
        string t;
        for(int i = 0; i < numRows; i++){
            for(int j = 0; j < 1001; j++){
                if(temp[i][j] == '0') continue;
                t += temp[i][j];
            }
        }
        return t;
    }
};
```
