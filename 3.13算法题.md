## [42. 接雨水 - 力扣（LeetCode）](https://leetcode.cn/problems/trapping-rain-water/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.13-1-2026-03-13-13%3A26%3A53.png)

```cpp
class Solution {
public:
    int trap(vector<int>& height) {
        int n = height.size();//记录数组大小
        if(n == 0){
            return 0;//数组为空，不能接雨水
        }
        vector<int> leftMax(n);//存储每个位置左边最大高度
        leftMax[0] = height[0];//第一个位置最大高度为自己
        for(int i = 1; i < n; ++i){
            leftMax[i] = max(leftMax[i - 1], height[i]);//记录左侧最大高度
        }
        vector<int> rightMax(n);//存储每个位置右边最大高度
        rightMax[n - 1] = height[n - 1];//右边最大高度为自己
        for(int i = n - 2; i >= 0; --i){
            rightMax[i] = max(rightMax[i + 1],height[i]);//记录右侧最大高度
        }
        int res = 0;//存储结果
        for(int i = 0; i < n; ++i){
            res += min(leftMax[i],rightMax[i]) - height[i];//每个位置接到的量
        }
        return res;
    }
};
```

## [3. 无重复字符的最长子串 - 力扣（LeetCode）](https://leetcode.cn/problems/longest-substring-without-repeating-characters/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.13-2-2026-03-13-13%3A50%3A04.png)

```cpp
class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        int n = s.size();//获取当前数组大小
        if( n == 0){
            return 0;//数组为0，最长为0
        }
        vector<int> Hash(256,-1);//建立哈希表
        int res = 0;//存储结果
        int l = 0;//左指针
        for(int r = 0; r < n; ++r){//右指针
            char t = s[r];//暂时存放右指针的字符
            if(Hash[t] >= l){//该字符已重复
                l = Hash[t] + 1;//跳过重复
            }
            Hash[t] = r;//右指针位置为该字符的值
            int tc = r - l + 1;//当前窗口长度
            res = max(tc,res);//更新窗口保证最大
        }
        return res;
    }
};
```
