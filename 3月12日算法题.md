## [45. 跳跃游戏 II - 力扣（LeetCode）](https://leetcode.cn/problems/jump-game-ii/)

![屏幕截图 2026-03-12 202113.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-03-12%20202113.png?raw=true)

```cpp
class Solution {
public:
    int jump(vector<int>& nums) {
        int nextIndex = 0; // 下一步覆盖最远距离下标
        int curIndex = 0;// 当前覆盖最远距离下标
        int n = nums.size();
        if(n == 1){
            return 0;
        }
        int count = 0;// 记录走的最大步数
        for(int i = 0; i < n; i++){
            nextIndex = max(nextIndex, i+nums[i]);// 更新下一步覆盖最远距离下标
            if(i == curIndex){ // 遇到当前覆盖最远距离下标
                curIndex = nextIndex; // 更新当前覆盖最远距离下标（相当于加油了）
                count++;// 需要走下一步
                if(nextIndex >= n-1){
                    break;
                }// 当前覆盖最远距到达集合终点，不用做count++操作了，直接结束
            }
            
        }
        return count;
    }
};
```

## [1005. K 次取反后最大化的数组和 - 力扣（LeetCode）](https://leetcode.cn/problems/maximize-sum-of-array-after-k-negations/description/)

![屏幕截图 2026-03-12 205106.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-03-12%20205106.png?raw=true)

```cpp
class Solution {
public:
    int largestSumAfterKNegations(vector<int>& nums, int k) {
        sort(nums.begin(), nums.end(), cmp);
        int n = nums.size();
        int sum = 0;
        //从后向前遍历，把遇到的负数变为正数，同时k--
        for(int i = n - 1; i >= 0; i--){
            if(nums[i] < 0 && k > 0){
                nums[i] *= -1;
                k--;
            }
        }
        //如果k还大于0，就反复转变绝对值最小的数，把k用完
        if(k > 0){
            if(k%2 != 0){
                nums[0] *= -1;
            }
        }
        //遍历数组求和
        for(int i = 0; i < n; i++){
            sum += nums[i];
        }
        return sum;
    }
    // 按照绝对值大小升序排序
    static bool cmp(int a, int b){
        return abs(a) < abs(b);
    }
};
```
