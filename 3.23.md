## [406. 根据身高重建队列](https://leetcode.cn/problems/queue-reconstruction-by-height/submissions/711117385/)

![image-20260323213646228](C:\Users\OMEN\AppData\Roaming\Typora\typora-user-images\image-20260323213646228.png)

```c
int cmp(const void *p1, const void *p2) {
    int* pp1 = *(int**)p1;
    int* pp2 = *(int**)p2;
    return pp1[0] == pp2[0] ? pp1[1] - pp2[1] : pp2[0] - pp1[0]; 
}

//将数组元素后移一位
void move(int** people, int start, int end) {
    for(int i = end; i > start; i--) {
        people[i] = people[i - 1];
    }
}

int** reconstructQueue(int** people, int peopleSize, int* peopleColSize, int* returnSize, int** returnColumnSizes) {
    qsort(people, peopleSize, sizeof(int*), cmp);  //先按身高排序
    for(int i = 0; i < peopleSize; i++) {
        int postion = people[i][1];
        int* temp = people[i];
        move(people, postion, i);
        people[postion] = temp;
    }

    *returnSize = peopleSize;
    *returnColumnSizes = (int*)malloc(sizeof(int*) * peopleSize);
    for(int i = 0; i < peopleSize; i++) {
        (*returnColumnSizes)[i] = 2;
    }

    return people;
}
```

int cmp(const void *a,const void *b) {
    return ((*(int**)a)[0] > (*(int**)b)[0]);
} 

int findMinArrowShots(int** points, int pointsSize, int* pointsColSize){
    //先排序
    qsort(points, pointsSize, sizeof(points[0]),cmp);
    
    int arrowNum = 1;
    int i = 1;
    for(i = 1; i < pointsSize; i++) {
        //若前一个与当前气球不重叠
        if(points[i][0] > points[i-1][1]) {
            arrowNum++;
        } else {
            //若重叠
            points[i][1] = points[i][1] > points[i-1][1] ? points[i-1][1] : points[i][1];
        }
    }
    return arrowNum;
}
