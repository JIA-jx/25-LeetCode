```c
int wiggleMaxLength(int* nums, int numsSize) {
    if(numsSize <= 1) {
        return numsSize;
    }

    int preDiff = 0, curDiff = 0;
    int res = 1;
    for(int i = 0; i < numsSize - 1; i++) {
        curDiff = nums[i + 1] - nums[i];
        if(preDiff <= 0 && curDiff > 0 || preDiff >= 0 && curDiff < 0) {
            res++;
            preDiff = curDiff;
        }
    }
    return res;
}
```

## [225. 用队列实现栈](https://leetcode.cn/problems/implement-stack-using-queues/submissions/716670460/)

![image-20260409195700471](C:\Users\OMEN\AppData\Roaming\Typora\typora-user-images\image-20260409195700471.png)

```c
typedef struct {
    int queue1[100], queue2[100];
    int front1, rear1, front2, rear2; 
} MyStack;


MyStack* myStackCreate() {
    MyStack* stack = (MyStack*)malloc(sizeof(MyStack));
    stack->front1 = stack->front2 = stack->rear1 = stack->rear2 = 0;
    return stack;
}

void myStackPush(MyStack* obj, int x) {
    obj->queue1[obj->rear1++] = x;
}

int myStackPop(MyStack* obj) {
    while(obj->front1 < obj->rear1 - 1) {
        obj->queue2[obj->rear2++] = obj->queue1[obj->front1++];
    }
    int x = obj->queue1[obj->front1++];
    while(obj->front2 < obj->rear2) {
        obj->queue1[obj->rear1++] = obj->queue2[obj->front2++];
    }
    return x;
}

int myStackTop(MyStack* obj) {
    return obj->queue1[obj->rear1 - 1];
}

bool myStackEmpty(MyStack* obj) {
    return obj->front1 == obj->rear1  && obj->front2 == obj->rear2;
}

void myStackFree(MyStack* obj) {
    obj->front1 = obj->front2 = obj->rear1 = obj->rear2 = 0;
}
```
