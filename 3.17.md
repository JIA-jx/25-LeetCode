## [47. 全排列 II](https://leetcode.cn/problems/permutations-ii/submissions/708508377/)

![image-20260317212047079](C:\Users\OMEN\AppData\Roaming\Typora\typora-user-images\image-20260317212047079.png)

```c
int* path;
int pathTop;
int** ans;
int ansTop;
int* used;

int cmp(const void *a, const void *b) {
    return *(int*)a - *(int*)b;
}

void backbreaking(int* nums, int numsSize) {
    if(pathTop == numsSize) {
        int* temp = (int*)malloc(sizeof(int) * pathTop);
        for(int j = 0; j < pathTop; j++) {
            temp[j] = path[j];
        }
        ans[ansTop++] = temp;
        return;
    }

    for(int i = 0; i < numsSize; i++) {
        if(i > 0 && nums[i] == nums[i - 1] && used[i - 1] == 0) {  //如果前一个元素和当前的相等且未被使用过，跳过
            continue;
        }
        if(used[i] == 1) {  //如果当前元素已经使用过了，跳过
            continue;
        }

        path[pathTop++] = nums[i];
        used[i] = 1;

        backbreaking(nums, numsSize);
		//回溯
        pathTop--;
        used[i] = 0;
    }
}

int** permuteUnique(int* nums, int numsSize, int* returnSize, int** returnColumnSizes) {
    path = (int*)malloc(sizeof(int) * 8);
    ans = (int**)malloc(sizeof(int*) * 100000);
    used = (int*)malloc(sizeof(int) * 8);

    pathTop = ansTop = 0;
    qsort(nums, numsSize, sizeof(int), cmp);  //排序，方便去重操作

    backbreaking(nums, numsSize);

    *returnSize = ansTop;
    *returnColumnSizes = (int*)malloc(sizeof(int) * ansTop);
    for(int i = 0; i < ansTop; i++) {
        (*returnColumnSizes)[i] = numsSize;
    }

    return ans;
}
```

## [51. N 皇后](https://leetcode.cn/problems/n-queens/description/)

![image-20260317225114666](C:\Users\OMEN\AppData\Roaming\Typora\typora-user-images\image-20260317225114666.png)

```c
char** path;
char*** ans;
int ansTop;

void copy(int n) {
    char** temp = (char**)malloc(sizeof(char*) * n);
    for(int i = 0; i < n; i++) {
        temp[i] = (char*)malloc(sizeof(char) * (n + 1));
        for(int j = 0; j < n; j++) {
            temp[i][j] = path[i][j];
        }
        temp[i][n] = '\0';
    }
    ans[ansTop++] = temp;
}

int isValid(int n, int x, int y) {  //判断该位置放皇后是否合法
    for(int i = 0; i < n; i++) {
        if(path[x][i] == 'Q' || path[i][y] == 'Q') {
            return 0;
        }
    }

    int i = x - 1;
    int j = y - 1;
    while(i >= 0 && j >= 0) {
        if(path[i][j] == 'Q') {
            return 0;
        }
        i--;
        j--;
    }

    i = x + 1;
    j = y + 1;
    while(i < n && j < n) {
        if(path[i][j] == 'Q') {
            return 0;
        }
        i++;
        j++;
    }

    i = x - 1;
    j = y + 1;
    while(i >= 0 && j < n) {
        if(path[i][j] == 'Q') {
            return 0;
        }
        i--;
        j++;
    }

    i = x + 1;
    j = y - 1;
    while(j >= 0 && i < n) {
        if(path[i][j] == 'Q') {
            return 0;
        }
        j--;
        i++;
    }
    return 1;
}

void initPath(int n) {  //将所有位置全初始化为'.'
    int i, j;
    for(i = 0; i < n; i++) {
        path[i] = (char*)malloc(sizeof(char) * n + 1);
        for(j = 0; j < n; j++)
            path[i][j] = '.';
        path[i][j] = '\0';
    }
}

void backtracking(char** path, int n, int row) {
    if(row == n) {
        copy(n);
        return;
    }

    for(int i = 0; i < n; i++) {
        if(isValid(n, row, i)) {
            path[row][i] = 'Q';
            backtracking(path, n, row + 1);
            path[row][i] = '.';
        } else {
            continue;
        }
    }
}

char*** solveNQueens(int n, int* returnSize, int** returnColumnSizes) {
    path = (char**)malloc(sizeof(char*) * n);
    ans = (char***)malloc(sizeof(char**) * 1000);

    ansTop = 0;
    initPath(n);
    backtracking(path, n, 0);

    *returnSize = ansTop;
    *returnColumnSizes = (int*)malloc(sizeof(int) * ansTop);
    for(int i = 0; i < ansTop; i++) {
        (*returnColumnSizes)[i] = n;
    }

    return ans;
}
```
