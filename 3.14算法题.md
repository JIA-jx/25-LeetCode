## [438. 找到字符串中所有字母异位词 - 力扣（LeetCode）](https://leetcode.cn/problems/find-all-anagrams-in-a-string/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.14-1-2026-03-14-22%3A23%3A32.png)

```cpp
class Solution {
public:
    vector<int> findAnagrams(string s, string p) {
        int slen = s.size();
        int plen = p.size();
        if(slen < plen){//不可能出现异位词
            return {};
        }
        vector<int> count (26,0);//初始化
        vector<int> res;
        for (char c : p) {
            count[c - 'a']++;//p出现的次数
        }
        int l = 0, r = 0,need = plen;
        while(r < slen){//右指针边界
            if(count[s[r] - 'a'] > 0) --need;//p中字符被出现了
            count[s[r] - 'a']--;//减少计数
            r++;//右指针右移
            if(r - l == plen){//窗口等于p长度
                if(need == 0){//符合条件
                    res.push_back(l);//添加结果
                }
                if(count[s[l] - 'a'] >=0) ++need;//左指针右移
                ++count[s[l] - 'a'];
                ++l;
            }
        }
        return res;
    }
};
```

## [560. 和为 K 的子数组 - 力扣（LeetCode）](https://leetcode.cn/problems/subarray-sum-equals-k/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.14-2-2026-03-14-23%3A07%3A14.png)

```cpp
class Solution {
public:
    int subarraySum(vector<int>& nums, int k) {
        unordered_map<int,int> mp;
        mp[0] = 1;//前缀和为0时出现1次
        int count = 0, pre = 0;
        for(auto& x : nums){
            pre += x;//更新前缀和
            if(mp.find(pre - k) != mp.end()){//pre- k存在于数组中
                count += mp[pre - k];出现k的次数
            }
            mp[pre]++;//pre出现次数
        }
        return count;
    }
};
```
