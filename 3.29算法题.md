## [105. 从前序与中序遍历序列构造二叉树 - 力扣（LeetCode）](https://leetcode.cn/problems/construct-binary-tree-from-preorder-and-inorder-traversal/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.29-1-2026-03-29-19%3A55%3A16.png)

```cpp
class Solution {
public:
    TreeNode* buildTree(vector<int>& preorder, vector<int>& inorder) {
        this->preorder = preorder;//递归函数使用
        for(int i = 0; i < inorder.size(); ++i)
            dic[inorder[i]] = i;//中序的值映射到其索引
            return recur(0, 0, inorder.size() - 1);//启动递归
    }
private:
    vector<int> preorder;
    unordered_map<int, int> dic;
    TreeNode* recur(int root, int left,int right){
        if(left >right) return nullptr;//空节点停止递归
        TreeNode* node = new TreeNode(preorder[root]);//创建子树根节点
        int i = dic[preorder[root]];//找到该节点在中序遍历中的位置，划分左右子树
        node->left = recur(root + 1, left, i - 1);//构建左子树
        node->right = recur(root + i - left + 1, i + 1, right);//构建右子树
        return node;//返回当前子树根节点
    }
};
```

## [437. 路径总和 III - 力扣（LeetCode）](https://leetcode.cn/problems/path-sum-iii/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.29-2-2026-03-29-20%3A22%3A11.png)

```cpp
class Solution {
public:
    int pathSum(TreeNode* root, int targetSum) {
        unordered_map<long long, int> cnt = {{0, 1}};//前缀和为0的情况出现一次
        int ans = 0;//答案
        auto dfs = [&](this auto&& dfs, TreeNode* node, long long s){//构造lambda表达式
            if(node == nullptr) return;//当节点为空停止递归
            s += node->val;//记录前缀和
            ans += cnt[s - targetSum];//有一个对应的targetSum与之对应
            cnt[s]++;//记录该前缀和中的正确数量
            dfs(node->left,s);//向左递归
            dfs(node->right,s);//向右递归
            cnt[s]--;//退出路径，清理记录
        };
        dfs(root, 0);//开始d
        return ans;
    }
};
```
