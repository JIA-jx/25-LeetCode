## [455. 分发饼干 - 力扣（LeetCode）](https://leetcode.cn/problems/assign-cookies/description/)

![屏幕截图 2026-03-11 193748.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-03-11%20193748.png?raw=true)

```cpp
class Solution {
public:
    int findContentChildren(vector<int>& g, vector<int>& s) {
        //排序
        sort(g.begin(), g.end());
        sort(s.begin(), s.end());
        int i = 0, j = 0, count = 0;
        while(i < g.size() && j < s.size()){
            //j饼干能够满足当前i孩子的胃口，count+1
            if(s[j] >= g[i]){
                i++;
                j++;
                count++;
            }
            //j饼干不能满足当前i孩子的胃口，就让j++，换更大的饼干继续和当前孩子的胃口比较
            else{
                j++;
            }
        }
        return count;
    }
};
```

## [376. 摆动序列 - 力扣（LeetCode）](https://leetcode.cn/problems/wiggle-subsequence/description/)

![屏幕截图 2026-03-11 203352.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-03-11%20203352.png?raw=true)

```cpp
class Solution {
public:
    int wiggleMaxLength(vector<int>& nums) {
        if(nums.size() <= 1){
            return nums.size();
        }
        int count = 1;
        int preDiff = 0;// 前一对差值
        int curDiff = 0;// 当前一对差值
        int n = nums.size();
        for(int i = 0; i < n - 1; i++){
            curDiff = nums[i+1] - nums[i];
             // 出现峰值
            if((preDiff <= 0 && curDiff > 0) || (preDiff >= 0 && curDiff < 0)){
                //正负变化时更新preDiff
                preDiff = curDiff;
                count++;
            }
        }
        return count;
    }
};
```
