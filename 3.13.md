## [40. 组合总和 II](https://leetcode.cn/problems/combination-sum-ii/submissions/706684641/)

![image-20260313185405487](C:\Users\OMEN\AppData\Roaming\Typora\typora-user-images\image-20260313185405487.png)

```c
int* path;
int pathTop;
int** ans;
int ansTop;
int* length;

int cmp(const void* a, const void* b) {
    return *((int*)a) - *((int*)b);
}

void backtracking(int* candidates, int candidatesSize, int target, int sum, int startIndex) {
    if(sum > target) {
        return;
    }
    if(sum == target) {
        int* temp = (int*)malloc(sizeof(int) * pathTop);
        for(int j = 0; j < pathTop; j++) {
            temp[j] = path[j];
        }
        length[ansTop] = pathTop;
        ans[ansTop++] = temp;
        return;
    }

    for(int i = startIndex; i < candidatesSize; i++) {
        if(i > startIndex && candidates[i] == candidates[i - 1]) {   
            //如果索引为i的数字和上一个相同，则为了不得到重复结果，直接跳过
            continue;
        }
        path[pathTop++] = candidates[i];
        sum += candidates[i];
        backtracking(candidates, candidatesSize, target, sum, i + 1);
        pathTop--;
        sum -= candidates[i];
    }
}

int** combinationSum2(int* candidates, int candidatesSize, int target, int* returnSize, int** returnColumnSizes) {
    path = (int*)malloc(sizeof(int) * 100);
    ans = (int**)malloc(sizeof(int*) * 100);
    length = (int*)malloc(sizeof(int) * 100);

    pathTop = ansTop = 0;

    qsort(candidates, candidatesSize, sizeof(int), cmp);  //对数组进行排序方便后续去重

    backtracking(candidates, candidatesSize, target, 0, 0);

    *returnSize = ansTop;
    *returnColumnSizes = (int*)malloc(sizeof(int) * ansTop);
    for(int i = 0; i < ansTop; i++) {
        (*returnColumnSizes)[i] = length[i];
    }

    return ans;
}
```



## [67. 二进制求和](https://leetcode.cn/problems/add-binary/submissions/706707203/)

![image-20260313195628439](3.13.assets/image-20260313195628439.png)

```c
void reverse(char* a) {   //反转字符串，末尾对齐
    int len = strlen(a);
    for(int i = 0; i < len / 2; i++) {
        char temp = a[i];
        a[i] = a[len - 1 - i];
        a[len - 1 - i] = temp;
    }
}

char* addBinary(char* a, char* b) {
    reverse(a);
    reverse(b);
    int n = (strlen(a) > strlen(b)) ? strlen(a) : strlen(b);
    char* x = (char*)malloc(sizeof(char) * (n + 2));
    int carry = 0;  //以carry表示上一位是否进位
    for(int i = 0; i < n; i++) {
        carry += (i < strlen(a)) ? (a[i] == '1') : 0;
        carry += (i < strlen(b)) ? (b[i] == '1') : 0;
        x[i] = carry % 2 + '0';
        carry /= 2;
    }
    if(carry) {
        x[n] = '1';
        x[n + 1] = '\0';
    } else {
        x[n] = '\0';
    }
    reverse(x);
    return x;
}
```
