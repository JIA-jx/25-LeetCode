## [739. 每日温度 - 力扣（LeetCode）](https://leetcode.cn/problems/daily-temperatures/)

![屏幕截图 2026-03-26 190514.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-03-26%20190514.png?raw=true)

```cpp
class Solution {
public:
    vector<int> dailyTemperatures(vector<int>& temperatures) {
        stack<int> nums;
        vector<int> result(temperatures.size(), 0);
		//遍历temperatures数组
        for(int i = 0; i < temperatures.size(); i++){
            //只要当前元素比栈顶元素大
            while(!nums.empty() && temperatures[i] > temperatures[nums.top()]){
                result[nums.top()] = i - nums.top();//计算栈顶元素result值
                nums.pop();//弹出栈顶元素
            }
            nums.push(i);//当前元素入栈
        }

        return result;
    }
};
```

## [496. 下一个更大元素 I - 力扣（LeetCode）](https://leetcode.cn/problems/next-greater-element-i/description/)

![屏幕截图 2026-03-26 200226.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-03-26%20200226.png?raw=true)

```cpp
class Solution {
public:
    vector<int> nextGreaterElement(vector<int>& nums1, vector<int>& nums2) {
        stack<int> st;
        unordered_map<int, int> hash;
        vector<int> result(nums1.size(), -1);
        
        //先处理nums2,在哈希表里标记nums2[j]的下一个最大元素的值
        for(int j = 0; j < nums2.size(); j++){
            //找到就加入哈希表
            while(!st.empty() && nums2[j] > st.top()){
                hash[st.top()] = nums2[j];
                st.pop();
            }
            
            st.push(nums2[j]);
        }

        //遍历nums1,通过哈希表key值找到key为nums1[i]的value
        for(int i = 0; i < nums1.size(); i++){
            result[i] = hash[nums1[i]] == 0 ? -1: hash[nums1[i]];//value为0说明没找到，赋为-1，否则付找到的值
        }

        return result;
    }
};
```
