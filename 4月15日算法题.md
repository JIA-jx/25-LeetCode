## [11. 盛最多水的容器 - 力扣（LeetCode）](https://leetcode.cn/problems/container-with-most-water/description/?envType=problem-list-v2&envId=two-pointers)

![屏幕截图 2026-04-15 195937.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-04-15%20195937.png?raw=true)

```cpp
class Solution {
public:
    int maxArea(vector<int>& height) {
        int l = 0;                  // 左指针
        int r = height.size() - 1;  // 右指针
        int max_w = 0;              // 最大容量

        while(l < r) {
            // 短板决定容量，计算并更新最大值
            if(height[l] <= height[r]) {
                max_w = max(max_w, height[l] * (r-l));
                l++;  // 移动左短板指针
            } else {
                max_w = max(max_w, height[r] * (r-l));
                r--;  // 移动右短板指针
            }
        }
        return max_w;
    }
};
```

## [16. 最接近的三数之和 - 力扣（LeetCode）](https://leetcode.cn/problems/3sum-closest/description/?envType=problem-list-v2&envId=two-pointers)

![屏幕截图 2026-04-15 204455.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-04-15%20204455.png?raw=true)

```cpp
class Solution {
public:
    int threeSumClosest(vector<int>& nums, int target) {
        sort(nums.begin(), nums.end()); // 排序，便于双指针
        int min_d = INT_MAX; // 记录最小差值

        // 枚举第一个数
        for(int i = 0; i < nums.size(); i++){
            int cur_t = target - nums[i];
            int l = i+1, r = nums.size()-1; // 双指针
            
            while(l < r){
                int cur_d = cur_t - nums[l] - nums[r]; // 当前差值
                if(cur_d == 0) return target; // 完全匹配，直接返回
                
                // 更新最小差值
                if(abs(cur_d) < abs(min_d)) min_d = cur_d;
                
                // 双指针移动
                if(cur_d > 0) l++;
                else r--;
            }
        }
        return target - min_d; // 计算最接近的和
    }
};
```
