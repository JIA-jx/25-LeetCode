## [17. 电话号码的字母组合 - 力扣（LeetCode）](https://leetcode.cn/problems/letter-combinations-of-a-phone-number/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/4.6-1-2026-04-06-21%3A17%3A38.png)

```cpp
class Solution {
    string MAPPING[10] = {"", "", "abc", "def", "ghi", "jkl", "mno", "pqrs", "tuv", "wxyz"};//映射0 - 9的字母
public:
    vector<string> letterCombinations(string digits) {
        int n = digits.size();//接收到的数字个数
        if(n == 0){
            return {};
        }
        vector<string> res;//记录结果
        string path(n,0);//
        auto dfs = [&](this auto&& dfs, int i)-> void{
            if(i == n){//输入完全部数字递归终止
                res.emplace_back(path);//放入结果
                return;
            }
            for(char c : MAPPING[digits[i] - '0']){//获取当前数字可能对应的字母
                path[i] = c;//放入可能的单词
                dfs(i + 1);//进行下一轮递归
            }
        };
        dfs (0);
        return res;
    }
};
```

## [22. 括号生成 - 力扣（LeetCode）](https://leetcode.cn/problems/generate-parentheses/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/4.6-2-2026-04-06-21%3A28%3A44.png)

```cpp
class Solution {
public:
    vector<string> generateParenthesis(int n) {
        vector<string> res;//记录结果
        vector<int> path;//记录左括号位置
        auto dfs = [&](this auto&& dfs, int i, int balance){
            if(path.size() == n){//放满n个左括号，递归终止
                string s(n * 2, ')');//全部放上右括号
                for(int j : path){//找到对应位置修改为左括号
                    s[j] = '(';
                }
                res.emplace_back(s);//放入结果
                return;
            }
            for(int right = 0; right <= balance; right++){
                path.push_back( i + right);//放置一个左括号位置
                dfs(i + right + 1, balance - right + 1);//递归到下一层
                path.pop_back();//恢复现场
            }
        };
        dfs(0, 0);//调用递归
        return res;
    }
};
```
