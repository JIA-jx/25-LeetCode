## [704. 二分查找 - 力扣（LeetCode）](https://leetcode.cn/problems/binary-search/)

![屏幕截图 2026-04-07 210254.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-04-07%20210254.png?raw=true)

```c
int search(int* nums, int numsSize, int target) {
    int left = 0, right = numsSize - 1, i = numsSize / 2; // 左右边界、中间索引
    while(left <= right){
        if(nums[i] == target){ // 找到目标
            return i;
        }
        else if(nums[i] > target){ // 目标在左区间
            right = i - 1;
            i = (right - left) / 2 + left;
        }
        else { // 目标在右区间
            left = i + 1;
            i = (right - left) / 2 + left;
        }
    }
    return -1; // 未找到
}
```

## [27. 移除元素 - 力扣（LeetCode）](https://leetcode.cn/problems/remove-element/)

![屏幕截图 2026-04-07 210309.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-04-07%20210309.png?raw=true)

```c
int removeElement(int* nums, int numsSize, int val) {
    int left = 0, right = numsSize - 1;
    // 单个元素且为目标值，直接返回0
    if(numsSize == 1 && nums[0] == val){
        return 0;
    }
    // 双指针覆盖移除元素
    while(left <= right){
        // 右指针为目标值，直接跳过
        if(nums[right] == val){
            right--;
            numsSize--;
            continue;
        }
        // 左指针为目标值，与右交换
        if(nums[left] == val){
            int temp = nums[right];
            nums[right] = nums[left];
            nums[left] = temp;
            numsSize--;
            right--;
        }
        left++;
    }
    return numsSize;
}
```
