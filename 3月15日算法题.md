## [134. 加油站 - 力扣（LeetCode）](https://leetcode.cn/problems/gas-station/solutions/488357/jia-you-zhan-by-leetcode-solution/)

```cpp
class Solution {
public:
    int canCompleteCircuit(vector<int>& gas, vector<int>& cost) {
        int curSum = 0;//从起始位置跑到当前位置剩余油量
        int totalSum = 0;//跑完全程剩余油量
        int start = 0;
        for(int i = 0; i < gas.size(); i++){
            int rest = gas[i] - cost[i];
            curSum += rest;
            totalSum += rest;
            if(curSum < 0){// 当前累加rest和 curSum一旦小于0
                start = i + 1; // 起始位置更新为i+1
                curSum = 0; // curSum从0开始
            }
        }
        if(totalSum < 0){// 说明怎么走都不可能跑一圈了
            return -1;
        }
        return start;
    }
};
```

## [122. 买卖股票的最佳时机 II - 力扣（LeetCode）](https://leetcode.cn/problems/best-time-to-buy-and-sell-stock-ii/description/)

![屏幕截图 2026-03-15 181112.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-03-15%20181112.png?raw=true)

```cpp
class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int n = prices.size();
        int sum = 0;
        int diff = 0;
        //收集所有每天的正利润，即可得到全局最大利润
        for(int i = 0; i < n - 1; i++){
            diff = prices[i+1] - prices[i];//每天的利润
            if(diff > 0){
                sum += diff;//正利润就收集
            }
        }
        return sum;
    }
};
```
