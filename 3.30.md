## [208. 实现 Trie (前缀树)](https://leetcode.cn/problems/implement-trie-prefix-tree/submissions/713656273/)

![image-20260330220601588](C:\Users\OMEN\AppData\Roaming\Typora\typora-user-images\image-20260330220601588.png)

```c
typedef struct Trie {
    struct Trie* children[26];
    bool isEnd;
} Trie;

Trie* trieCreate() {
    Trie* ret = malloc(sizeof(Trie));
    memset(ret->children, 0, sizeof(ret->children));
    ret->isEnd = false;
    return ret;
}

void trieInsert(Trie* obj, char* word) {
    int n = strlen(word);
    for (int i = 0; i < n; i++) {
        int ch = word[i] - 'a';
        if (obj->children[ch] == NULL) {
            obj->children[ch] = trieCreate();
        }
        obj = obj->children[ch];
    }
    obj->isEnd = true;
}

bool trieSearch(Trie* obj, char* word) {
    int n = strlen(word);
    for (int i = 0; i < n; i++) {
        int ch = word[i] - 'a';
        if (obj->children[ch] == NULL) {
            return false;
        }
        obj = obj->children[ch];
    }
    return obj->isEnd;
}

bool trieStartsWith(Trie* obj, char* prefix) {
    int n = strlen(prefix);
    for (int i = 0; i < n; i++) {
        int ch = prefix[i] - 'a';
        if (obj->children[ch] == NULL) {
            return false;
        }
        obj = obj->children[ch];
    }
    return true;
}

void trieFree(Trie* obj) {
    for (int i = 0; i < 26; i++) {
        if (obj->children[i]) {
            trieFree(obj->children[i]);
        }
    }
    free(obj);
}
```



## [518. 零钱兑换 II](https://leetcode.cn/problems/coin-change-ii/)

![image-20260330223024737](C:\Users\OMEN\AppData\Roaming\Typora\typora-user-images\image-20260330223024737.png)

```c
int change(int amount, int* coins, int coinsSize) {
    unsigned long long dp[amount + 1];
    memset(dp, 0, sizeof (dp));
    dp[0] = 1;
    for(int i = 0; i < coinsSize; i++){
        for(int j = coins[i]; j <= amount; j++){
            dp[j] += dp[j - coins[i]];
        }
    }
    return dp[amount];
}
```
