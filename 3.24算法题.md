## [2906. 构造乘积矩阵 - 力扣（LeetCode）](https://leetcode.cn/problems/construct-product-matrix/description/?envType=daily-question&envId=2026-03-24)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.24-1-2026-03-24-21%3A01%3A22.png)

```cpp
class Solution {
public:
    vector<vector<int>> constructProductMatrix(vector<vector<int>>& grid) {
        int mod = 12345;//题目要求取余数
        int n = grid.size(), m = grid[0].size();
        vector<vector<int>> p (n,vector<int>(m));//记录结果
        long long fix = 1;//定义后缀乘积
        for(int i = n - 1; i >= 0; --i){
            for(int j = m - 1; j >= 0; --j){
                p[i][j] = fix;//当前结果为后缀乘积
                fix = fix * grid[i][j] % mod;//更新后缀乘积
            }
        }
        long long pre = 1;//定义前缀乘积
        for(int i = 0; i < n; ++i){
            for(int j = 0; j < m; ++j){
                p[i][j] = p[i][j] * pre % mod;//当前结果为前缀乘积*后缀乘积取模
                pre = pre * grid[i][j] % mod;//更新前缀乘积
            }
        }
        return p;
    }
};
```

## [19. 删除链表的倒数第 N 个结点 - 力扣（LeetCode）](https://leetcode.cn/problems/remove-nth-node-from-end-of-list/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.24-2-2026-03-24-21%3A22%3A33.png)

```cpp
class Solution {
public:
    ListNode* removeNthFromEnd(ListNode* head, int n) {
        ListNode dummy {0, head};//创建哑节点
        ListNode* left = &dummy;//左指针位置
        ListNode* right = &dummy;//右指针位置
        while(n--){
            right = right->next;//让左右指针产生n个差值
        }
        while(right->next){//同时移动左右指针
            left = left->next;
            right = right->next;
        }
        ListNode* t = left->next;//t为被删除节点
        left->next = left->next->next;//跳过需要被删除节点
        delete t;//删除被删除节点
        return dummy.next;
    }
};
```
