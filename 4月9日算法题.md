## [416. 分割等和子集 - 力扣（LeetCode）](https://leetcode.cn/problems/partition-equal-subset-sum/description/)

![屏幕截图 2026-03-19 111110.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-03-19%20111110.png?raw=true)

```cpp
class Solution {
public:
    bool canPartition(vector<int>& nums) {
        int sum = 0;
        for(int i = 0; i < nums.size(); i++){
            sum += nums[i];
        }
        
        //如果数组和为奇数，说明不可能分割为两个元素和相等的子集
        if(sum % 2 == 1){
            return false;
        }
        int target = sum/2;

        vector<int> dp(target + 1, 0);
		
        //遍历数组每个数字
        for(int i = 0; i < nums.size(); i++){
            //逆序遍历容量，避免同一物品被重复选取
            for(int j = target; j >= nums[i]; j--){
                //获取选与不选该数字时的最大目标和
                dp[j] = max(dp[j], dp[j-nums[i]] + nums[i]);
            }
        }
        if(dp[target] == target){
            return true;
        }
        return false;
    }
};
```

## [1049. 最后一块石头的重量 II - 力扣（LeetCode）](https://leetcode.cn/problems/last-stone-weight-ii/description/)

![屏幕截图 2026-03-19 114837.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-03-19%20114837.png?raw=true)

```cpp
class Solution {
public:
    int lastStoneWeightII(vector<int>& stones) {
        int sum = 0;
        // 计算石头总重量
        for(int i = 0; i < stones.size(); i++){
            sum += stones[i];
        }
        int target = sum/2;
        // dp[j]：能凑出的≤j的最大子集和
        vector<int> dp(target+1, 0);
        for(int i = 0; i < stones.size(); i++){
             // 逆序遍历：避免同一石头被重复选取
            for(int j = target; j >= stones[i]; j--){
                dp[j] = max(dp[j], dp[j-stones[i]] + stones[i]);//选/不选当前石头，取最大值
            }
        }

        return sum - 2 * dp[target];
    }
};
```
