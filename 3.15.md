```c
/**
 * Return an array of arrays of size *returnSize.
 * The sizes of the arrays are returned as *returnColumnSizes array.
 * Note: Both returned array and *columnSizes array must be malloced, assume caller calls free().
 */
int* path;
int pathTop;
int** ans;
int ansTop;
int* length;

void backracking(int* nums, int numsSize, int startIndex) {
    //因为要收集所有子集，所以先收集结果
    int* temp = (int*)malloc(sizeof(int) * pathTop);
    for(int j = 0; j < pathTop; j++) {
        temp[j] = path[j];
    }
    length[ansTop] = pathTop;
    ans[ansTop++] = temp;
	//结束条件
    if(startIndex >= numsSize) {
        return;
    }

    for(int i = startIndex; i < numsSize; i++) {
        path[pathTop++] = nums[i];
        
        backracking(nums, numsSize, i + 1);

        pathTop--;
    }
}

int** subsets(int* nums, int numsSize, int* returnSize, int** returnColumnSizes) {
    path = (int*)malloc(sizeof(int) * 10);
    ans = (int**)malloc(sizeof(int*) * 10000);
    *returnColumnSizes = (int*)malloc(sizeof(int) * 10000);
    length = (int*)malloc(sizeof(int) * 10000);

    pathTop = 0;
    ansTop = 0;
    backracking(nums, numsSize, 0);

    *returnSize = ansTop;
    for(int i = 0; i < ansTop; i++) {
        (*returnColumnSizes)[i] = length[i];
    }

    return ans;
}
```

## [90. 子集 II](https://leetcode.cn/problems/subsets-ii/submissions/707567014/)

![image-20260315214334764](3.15.assets/image-20260315214334764.png)

```c
int* path;
int pathTop;
int** ans;
int ansTop;
int* length;

int cmp(const void *a, const void *b) {
    return *(int*)a - *(int*)b;
}

void backtracking(int* nums, int numsSize, int startIndex) {
    if(startIndex >= numsSize) {
        return;
    }

    for(int i = startIndex; i < numsSize; i++) {
        if(i > startIndex && nums[i] == nums[i - 1]) {  //同一层递归遇到重复元素时就跳过
            continue;
        } else{
            path[pathTop++] = nums[i];
            //收集结果
            int* temp = (int*)malloc(sizeof(int) * pathTop);
            for(int j = 0; j < pathTop; j++) {
                temp[j] = path[j];
            }
            length[ansTop] = pathTop;
            ans[ansTop++] = temp;

            backtracking(nums, numsSize, i + 1);
			//回溯
            pathTop--;
        }
    }
}

int** subsetsWithDup(int* nums, int numsSize, int* returnSize, int** returnColumnSizes) {
    path = (int*)malloc(sizeof(int) * 10);
    ans = (int**)malloc(sizeof(int*) * 10000);
    *returnColumnSizes = (int*)malloc(sizeof(int) * 10000);
    length = (int*)malloc(sizeof(int) * 10000);

    qsort(nums, numsSize, sizeof(int), cmp);  //先给各元素进行排序，方便后续去重

    length[0] = 0;
    pathTop = 0;
    ansTop = 1;

    backtracking(nums, numsSize, 0);

    *returnSize = ansTop;
    for(int i = 0; i < ansTop; i++) {
        (*returnColumnSizes)[i] = length[i];
    }

    return ans;
}
```
