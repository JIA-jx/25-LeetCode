## [455. 分发饼干](https://leetcode.cn/problems/assign-cookies/)

```c
int cmp(const void* a, const void* b) {
    return *(int*)a - *(int*)b;
}

int findContentChildren(int* g, int gSize, int* s, int sSize) {
    qsort(g, gSize, sizeof(int), cmp);
    qsort(s, sSize, sizeof(int), cmp);

    int num = 0;
    int j = sSize - 1;
    for(int i = gSize - 1; i >= 0; i--) {
        if(j >= 0 && s[j] >= g[i]) {
            num++;
            j--;
        }
    }
    return num;
}
```



## [39. 组合总和 ](https://leetcode.cn/problems/combination-sum/description/)

![image-20260408215239924](C:\Users\OMEN\AppData\Roaming\Typora\typora-user-images\image-20260408215239924.png)

```c
int* path;
int pathTop;
int** res;
int resTop;
int sum;
int* length;

void backtracking(int* candidates, int candidatesSize, int target, int sum, int index) {
    if(sum > target) {
        return;
    }

    if(sum == target) {
        int* temp = (int*)malloc(sizeof(int) * pathTop);
        for(int j = 0; j < pathTop; j++) {
            temp[j] = path[j];
        }
        length[resTop] = pathTop;
        res[resTop++] = temp;
        
        return;
    }

    for(int i = index; i < candidatesSize; i++) {
        path[pathTop++] = candidates[i];
        sum += candidates[i];
        backtracking(candidates, candidatesSize, target, sum, i);
        pathTop--;
        sum -= candidates[i];
    }
}

int** combinationSum(int* candidates, int candidatesSize, int target, int* returnSize, int** returnColumnSizes) {
    path = (int*)malloc(sizeof(int) * 40);
    res = (int**)malloc(sizeof(int*) * 200);
    length = (int*)malloc(sizeof(int) * 200);

    pathTop = resTop = 0;
    backtracking(candidates, candidatesSize, target, 0, 0);

    *returnSize = resTop;
    *returnColumnSizes = (int*)malloc(sizeof(int) * (*returnSize));
    for(int i = 0; i < resTop; i++) {
        (*returnColumnSizes)[i] = length[i];
    }

    return res;
}
```
