```c
int monotoneIncreasingDigits(int n) {
    char str[11];
    // 将数字转换为字符串
    sprintf(str, "%d", n);
    int len = strlen(str);
    int flag = strlen(str);
    for(int i = len - 1; i > 0; i--){
        if(str[i] < str[i - 1]){
            str[i - 1]--;
            flag = i;
        }
    }
    for(int i = flag; i < len; i++){
        str[i] = '9';
    }
    // 字符串转数字
    return atoi(str);
}
```

## [56. 合并区间](https://leetcode.cn/problems/merge-intervals/submissions/712016831/)

![image-20260325225820790](C:\Users\OMEN\AppData\Roaming\Typora\typora-user-images\image-20260325225820790.png)

```c
#define max(a, b) ((a) > (b) ? (a) : (b))

// 按左边界从小到大进行排序
int cmp(const void * var1, const void * var2){
    int *v1 = *(int **) var1;
    int *v2 = *(int **) var2;
    return v1[0] - v2[0];
}

int** merge(int** intervals, int intervalsSize, int* intervalsColSize, int* returnSize, int** returnColumnSizes) {
    int ** result = malloc(sizeof (int *) * intervalsSize);
    * returnColumnSizes = malloc(sizeof (int ) * intervalsSize);
    for(int i = 0; i < intervalsSize; i++){
        result[i] = malloc(sizeof (int ) * 2);
    }
    qsort(intervals, intervalsSize, sizeof (int *), cmp);
    int count = 0;
    for(int i = 0; i < intervalsSize; i++){
        // 记录区间的左右边界
        int L = intervals[i][0], R = intervals[i][1];
        // 如果count为0或者前一区间的右区间小于此时的左边
        if (count == 0 || result[count - 1][1] < L) {
            returnColumnSizes[0][count] = 2;
            result[count][0] = L;
            result[count][1] = R;
            count++;
        } 
        else{ // 更新右边界的值
            result[count - 1][1] = max(R, result[count - 1][1]);
        }
    }
    *returnSize = count;
    return result;
}
```
