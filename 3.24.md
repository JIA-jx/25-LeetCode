## [435. 无重叠区间](https://leetcode.cn/problems/non-overlapping-intervals/submissions/711291306/)

![image-20260324115328135](C:\Users\OMEN\AppData\Roaming\Typora\typora-user-images\image-20260324115328135.png)

```c
方法一：按左边界从小到大排序
int cmp(const void * a, const void *b) {
    return (*(int**)a)[0] > (*(int**)b)[0];
}

int eraseOverlapIntervals(int** intervals, int intervalsSize, int* intervalsColSize) {
    if(intervalsSize <= 1) {
        return 0;
    }

    qsort(intervals, intervalsSize, sizeof(int*), cmp);  //按左边界从小到大排序
    int res = 0;
    for(int i = 1; i < intervalsSize; i++) {
        if(intervals[i][0] < intervals[i - 1][1]) {  //若前一个的右边界与当前的左边界重叠，则将移除数量res + 1， 并将当前右边界更改为更小的那个（相当于移除右边界更大的区间）
            res++;
            intervals[i][1] = intervals[i - 1][1] < intervals[i][1] ? intervals[i - 1][1] : intervals[i][1];
        }
    }
    return res;
}
```

```c
方法二：按右边界从小到大排序
int cmp(const void *a, const void *b) {
    return (*(int**)a)[1] - (*(int**)b)[1];
}

int eraseOverlapIntervals(int** intervals, int intervalsSize, int* intervalsColSize) {
    if(intervalsSize <= 1) {
        return 0;
    }
    qsort(intervals, intervalsSize, sizeof(int*), cmp);
    int res = 0;
    for(int i = 1; i < intervalsSize; i++) {
        if(intervals[i - 1][1] > intervals[i][0]) {
            res++;
            intervals[i][1] = intervals[i - 1][1] < intervals[i][1] ? intervals[i - 1][1] : intervals[i][1];
        }
    }
    return res;
}
```



## [763. 划分字母区间](https://leetcode.cn/problems/partition-labels/submissions/711502413/)

![image-20260324194840684](3.24.assets/image-20260324194840684.png)

```c
int* partitionLabels(char* s, int* returnSize) {
    int hash[26] = {0};  //记录每个字母最后一次出现的索引
    for(int i = 0; i < strlen(s); i++) {
        hash[s[i] - 'a'] = i;
    }

    int* res = (int*)malloc(sizeof(int) * strlen(s));
    *returnSize = 0;
    int left = 0, right = 0;
    for(int i = 0; i < strlen(s); i++) {
        if(hash[s[i] - 'a'] > right) {  //找到该段子区间最大值作为右边界
            right = hash[s[i] - 'a'];
        }
        if(i == right) {  //当遍历到右边界时，记录长度，更新左边界
            res[(*returnSize)++] = right - left + 1;
            left = right + 1;
        }
    }
    return res;
}
```
