## [230. 二叉搜索树中第 K 小的元素 - 力扣（LeetCode）](https://leetcode.cn/problems/kth-smallest-element-in-a-bst/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.26-1-2026-03-26-21%3A26%3A34.png)

```cpp
class Solution {
public:
    int kthSmallest(TreeNode* root, int k) {
        this->k = k;//使k能够赋进dfs并改变k
         dfs(root);
         return res;
    }
private:
    int res,k;
    void dfs(TreeNode* root){
        if(!root) return;//节点不存在，递归停止
        dfs(root->left);//向左递归，因为左边值小于根
        if(k == 0) return;//剪枝，已寻找到
        if(--k == 0) res = root->val;//第k小的值
        dfs(root->right);//左递归结束未找到第k小的值，向右递归
    }
};
```

## [199. 二叉树的右视图 - 力扣（LeetCode）](https://leetcode.cn/problems/binary-tree-right-side-view/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.26-2-2026-03-26-21%3A40%3A59.png)

```cpp
class Solution {
public:
    vector<int> rightSideView(TreeNode* root) {
        vector<int> res;//记录结果
        dfs(root,0,res);//调用dfs
        return res;
    }
private:
    void dfs(TreeNode* node, int depth,vector<int>& res){
        if(!node){//节点为空，递归终止
            return;
        }
        if(depth == res.size()){//第一次到达该深度
            res.push_back(node->val);
        }
        dfs(node->right,depth + 1, res);//先向右遍历
        dfs(node->left,depth + 1, res);//再向左遍历
    }
};
```
