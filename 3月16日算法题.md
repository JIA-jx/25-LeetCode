## [135. 分发糖果 - 力扣（LeetCode）](https://leetcode.cn/problems/candy/description/)

![屏幕截图 2026-03-16 192302.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-03-16%20192302.png?raw=true)

```cpp
class Solution {
public:
    int candy(vector<int>& ratings) {
        int n = ratings.size();
        vector<int> result(n, 1); //初始化：每个孩子至少1颗糖果，满足基础条件
        
        //第一次遍历：从前到后，处理“右>左”的约束
        for(int i = 1; i < n; i++){
            //如果当前孩子评分比左边高，糖果数必须比左边多1
            if(ratings[i] > ratings[i-1]){
                result[i] = result[i-1] + 1;
            }
        }
        
        //第二次遍历：从后到前，处理“左>右”的约束
        for(int i = n-2; i >= 0; i--){
            //如果当前孩子评分比右边高，糖果数要取“当前值”和“右边+1”的最大值
            if(ratings[i] > ratings[i+1]){
                result[i] = max(result[i], result[i+1] + 1);
            }
        }
        int sum = 0;
        for(int i = 0; i < n; i++){
            sum += result[i];
        }
        return sum;
    }
};
```



## [860. 柠檬水找零 - 力扣（LeetCode）](https://leetcode.cn/problems/lemonade-change/description/)

![屏幕截图 2026-03-16 201219.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-03-16%20201219.png?raw=true)

```cpp
class Solution {
public:
    bool lemonadeChange(vector<int>& bills) {
        int fiveDollor = 0;//5美元的数量
        int tenDollor = 0;//10美元的数量
        
        int n = bills.size();
        if(bills[0] != 5){
            return false;
        }

        for(int i = 0; i < n; i++){
            //情况1
            if(bills[i] == 5){
                fiveDollor++;
            }
            
            //情况2
            else if(bills[i] == 10){
                if(fiveDollor == 0){
                    return false;
                }
                fiveDollor--;
                tenDollor++;
            }
            
            //情况3
            else if(bills[i] == 20){
                //先找出10美元
                if(tenDollor > 0){
                    if(fiveDollor == 0){
                        return false;
                    }
                    tenDollor--;
                    fiveDollor--;
                }
                //没有10美元再用5美元
                else{
                    if(fiveDollor < 3){
                        return false;
                    }
                    fiveDollor -= 3;
                }
            }
        }
        return true;
    }
};
```
