## [474. 一和零 - 力扣（LeetCode）](https://leetcode.cn/problems/ones-and-zeroes/)

![屏幕截图 2026-04-06 204358.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-04-06%20204358.png?raw=true)

```cpp
class Solution {
public:
    // 二维01背包：m个0、n个1，求最大子集长度
    int findMaxForm(vector<string>& strs, int m, int n) {
        // dp[j][k]：j个0、k个1的最大子集长度
        vector<vector<int>> dp(m+1, vector<int>(n+1, 0));
        
        for(int i = 0; i < strs.size(); i++){
            // 统计当前字符串的0、1数量
            int zeroNum = 0, oneNum = 0;
            for(char c : strs[i]){
                c == '0' ? zeroNum++ : oneNum++;
            }

            // 01背包：逆序遍历，防止重复选取
            for(int j = m; j >= zeroNum; j--){
                for(int k = n; k >= oneNum; k--){
                    // 状态转移：不选/选当前字符串
                    dp[j][k] = max(dp[j][k], dp[j-zeroNum][k-oneNum] + 1);
                }
            }
        }
        return dp[m][n];
    }
};
```

## [518. 零钱兑换 II - 力扣（LeetCode）](https://leetcode.cn/problems/coin-change-ii/)

```cpp
class Solution {
public:
    // 完全背包：求凑成amount的硬币组合数
    int change(int amount, vector<int>& coins) {
        // dp[j]：凑成金额j的组合数
        vector<int> dp(amount + 1, 0);
        // 初始化：金额0，组合数为1
        dp[0] = 1;

        // 遍历硬币（物品）
        for(int i = 0; i < coins.size(); i++){
            // 正序遍历：完全背包，可重复选
            for(int j = coins[i]; j <= amount; j++){
                // 防溢出 + 状态转移：累加组合数
                if(dp[j] < INT_MAX - dp[j - coins[i]]) {
                    dp[j] += dp[j-coins[i]];
                }
            }
        }
        return dp[amount];
    }
};
```
