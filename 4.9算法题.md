## [74. 搜索二维矩阵 - 力扣（LeetCode）](https://leetcode.cn/problems/search-a-2d-matrix/submissions/716688438/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/4.9-1-2026-04-09-20%3A55%3A39.png)

```cpp
class Solution {
public:
    bool searchMatrix(vector<vector<int>>& matrix, int target) {
        int m = 0, n = matrix[0].size() - 1;//将矩阵看成搜索二叉树
        while(m < matrix.size() && n >= 0){
            if(matrix[m][n] < target){
                m++;
            }
            else if(matrix[m][n] > target){
                n--;
            }
            else{
                return true;
            }
        }
        return false;
    }
};
```

## [79. 单词搜索 - 力扣（LeetCode）](https://leetcode.cn/problems/word-search/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/4.9-2-2026-04-09-20%3A59%3A10.png)

```cpp
class Solution {
public:
    bool exist(vector<vector<char>>& board, string word) {
        rows = board.size();
        cols = board[0].size();
        for(int i = 0; i < rows; i++){//遍历每行每类以寻找正确值
            for(int j = 0; j < cols; j++){
                if(dfs(board,word,i,j,0)) return true;
            }
        }
        return false;
    }
private:
    int rows, cols;
    bool dfs(vector<vector<char>>& board,string word,int i, int j, int k){
        if(i >= rows || i < 0 || j >= cols || j < 0 || board[i][j] != word[k]) return false;//不匹配
        if(k == word.size() - 1) return true;//找到正确单词
        board[i][j] = '\0';//标记为已被访问
        bool res = dfs(board,word,i + 1, j, k+1) || dfs(board,word,i - 1, j, k + 1) || dfs(board, word, i, j + 1, k + 1) || dfs(board, word, i , j - 1, k + 1);//向四周延申
        board[i][j] = word[k];//恢复现场，以便于下次使用
        return res;
    }
};
```
