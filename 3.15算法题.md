## [239. 滑动窗口最大值 - 力扣（LeetCode）](https://leetcode.cn/problems/sliding-window-maximum/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.15-1-2026-03-15-22%3A36%3A02.png)

```cpp
class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        vector<int> res;//存储最终结果
        deque<int> dq;//双端队列，存储元素索引，确保队列中元素对应的值递减
        int n = nums.size();//数组长度
        for(int r = 0; r < n; ++r){//r是窗口右边界，从0遍历到n - 1
            while(!dq.empty() && nums[r] >= nums[dq.back()]){
                dq.pop_back();//移除队列尾部所有小于等于当前元素的索引
            }
            dq.push_back(r);//将当前元素索引加入队列尾部
            while(dq.front() <= r - k){
                dq.pop_front();//移除队头超出窗口左边界索引
            }
            if(r >= k -1){//窗口大小达到k时
                res.push_back(nums[dq.front()]);//队头元素即当前窗口的最大值
            }
        }
        return res;
    }
};
```

## [76. 最小覆盖子串 - 力扣（LeetCode）](https://leetcode.cn/problems/minimum-window-substring/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.15-2-2026-03-15-23%3A18%3A05.png)

```cpp
class Solution {
public:
unordered_map<char, int> ori, cnt;//目标字符串中的出现次数，窗口中出现次数
    string minWindow(string s, string t) {
        for(const auto &c: t){
            ++ori[c];//统计目标字符次数
        }
        int l = 0, r = -1;//左右窗口
        int len = INT_MAX, ansL = -1;//最小子串长度，左边界
        while(r < int (s.size())){//r到右边界
            if(ori.find(s[++r]) != ori.end()){//向右移动窗口，且在目标字符串中
                ++cnt[s[r]];//当前窗口出现目标字符
            }
            while(check() && l <= r){//窗口满足条件
                if(r - l + 1 < len){//窗口长度小于最小长度
                    len = r - l + 1;//更新长度
                    ansL = l;//更新最小左边界
                }
                if(ori.find(s[l]) != ori.end()){//左边界是目标字符
                    --cnt[s[l]];//减少目前窗口中该字符数量
                }
                ++l;//右移左边界
            }
        }
        return ansL == -1? string() : s.substr(ansL,len);
    }
private:
    bool check() {//判断窗口满足条件
        for (const auto &p: ori) {
            if (cnt[p.first] < p.second) {
                return false;
            }
        }
        return true;
    }
};
```
