## [509. 斐波那契数 - 力扣（LeetCode）](https://leetcode.cn/problems/fibonacci-number/description/)

![屏幕截图 2026-03-18 193808.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-03-18%20193808.png?raw=true)

```cpp
class Solution {
public:
    int fib(int n) {
        if (n <= 1) return n;
        vector<int> dp(n + 1);
        dp[0] = 0;
        dp[1] = 1;
        //记录从0到n的所有斐波那契数
        for(int i = 2; i <= n; i++){
            dp[i] = dp[i-1] + dp[i-2];
        }
        return dp[n];
    }
};
```

## [70. 爬楼梯 - 力扣（LeetCode）](https://leetcode.cn/problems/climbing-stairs/description/)

![屏幕截图 2026-03-18 195935.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-03-18%20195935.png?raw=true)

```cpp
class Solution {
public:
    int climbStairs(int n) {
        if(n == 1){
            return 1;
        }
        int a = 1;
        int b = 2;
        
        for(int i = 3; i <= n; i++){
            int sum = a + b;//爬第i个楼梯的方法数量
            a = b;
            b = sum;
        }

        return b;
    }
};
```
