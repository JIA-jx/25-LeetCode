## [207. 课程表 - 力扣（LeetCode）](https://leetcode.cn/problems/course-schedule/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.30-1-2026-03-30-20%3A58%3A36.png)

```cpp
class Solution {
public:
    bool canFinish(int numCourses, vector<vector<int>>& prerequisites) {
        vector<int> indegrees(numCourses, 0);//统计度的值
        vector<vector<int>> adjacency(numCourses);//建立邻接表
        queue<int> q;//暂时存储度为0的值
        for(auto& cp : prerequisites){
            indegrees[cp[0]]++;//让前一个度加一
            adjacency[cp[1]].push_back(cp[0]);//让后一个值接在前一个值里
        }
        for(int i = 0; i < numCourses; ++i){
            if(indegrees[i] == 0){//当度为0，放入队列，方便进行bfs
                q.push(i);
            }
        }
        while(!q.empty()){//bfs
            int pre = q.front();//处理当前队列中的值
            q.pop();
            numCourses--;//学完一门课
            for(int cur : adjacency[pre]){//目前队列中可以让度减一的值
                    if(--indegrees[cur] == 0){//当该点度为0，可入队
                        q.push(cur);
                }
            }
        }
        return numCourses == 0;
    }
};
```

## [994. 腐烂的橘子 - 力扣（LeetCode）](https://leetcode.cn/problems/rotting-oranges/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.30-2-2026-03-30-21%3A47%3A04.png)

```cpp
class Solution {
    int DIRECTIONS[4][2] = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};//定义方向便于广搜
public:
    int orangesRotting(vector<vector<int>>& grid) {
        int m = grid.size(), n = grid[0].size();
        int fresh = 0;//记录新鲜水果数量
        vector<pair<int, int>> q;//储存腐烂橘子
        for(int i = 0; i < m; ++i){
            for(int j = 0; j < n; ++j){
                if(grid[i][j] == 1){//新鲜个数加一
                    fresh++;
                }else if(grid[i][j] == 2){//记录腐烂坐标
                    q.emplace_back(i,j);
                }
            }
        }
        int res = 0;//时间
        while(fresh && ! q.empty()){//当存在腐烂水果并且有新鲜水果
            res++;//经过一分钟
            vector<pair<int,int>> nxt;//存储下一时刻的腐烂水果
            for(auto& [x, y]: q){//从q中找到腐烂坐标
                for(auto d : DIRECTIONS){//四个方向开始传递
                    int i = x + d[0],j = y + d[1];//x、y坐标下一时刻的位置
                    if(0 <= i && i < m && 0 <= j && j < n && grid[i][j] == 1){//当前有新鲜水果
                        fresh--;//新鲜水果减1
                        grid[i][j] = 2;//标记为腐烂水果
                        nxt.emplace_back(i,j);//将该位置记录为腐烂水果
                    }
                }
            }
            q = move(nxt);//让q真正进入下一分钟
        }
        return fresh ? -1 : res;
    }
};
```
