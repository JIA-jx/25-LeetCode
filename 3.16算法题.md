## [53. 最大子数组和 - 力扣（LeetCode）](https://leetcode.cn/problems/maximum-subarray/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.16-12026-03-16-17%3A29%3A14.png)

```cpp
class Solution {
public:
    int maxSubArray(vector<int>& nums) {
        int res = nums[0];//结果为n个元素中最大和连续子串
        for(int i = 1; i < nums.size(); ++i){
            if(nums[i - 1] > 0) nums[i] += nums[i - 1];//当前子串为正，加上目前子串可以实现结果的增大
            if(nums[i] > res) res = nums[i];//更新结果值
        }
        return res;
    }
};
```

## [56. 合并区间 - 力扣（LeetCode）](https://leetcode.cn/problems/merge-intervals/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.16-2-2026-03-16-18%3A19%3A18.png)

```cpp
class Solution {
public:
    vector<vector<int>> merge(vector<vector<int>>& intervals) {
        if (intervals.size() == 0){//传入空值
            return {};//结果为空
        }
        vector<vector<int>> res;
        sort(intervals.begin(),intervals.end());//对每个区间进行排列
        for(int i = 0; i < intervals.size(); ++i){
            int L = intervals[i][0], R = intervals[i][1];//记录当前区间中左右边界
            if(!res.size() || res.back()[1] < L){//结果中未存放区间，结果的右区间小于当前左边界
                res.push_back({L,R});//压入新的区间
            }
            else{
                res.back()[1] = max(res.back()[1], R);//更新右边界
            }
        }
        return res;
    }
};
```
