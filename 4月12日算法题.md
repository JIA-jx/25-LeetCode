## [198. 打家劫舍 - 力扣（LeetCode）](https://leetcode.cn/problems/house-robber/)

![屏幕截图 2026-03-25 214008.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-03-25%20214008.png?raw=true)

```cpp
class Solution {
public:
    int rob(vector<int>& nums) {
        // 只有1间房，直接偷
        if(nums.size() == 1) return nums[0];
        // 只有2间房，偷金额大的
        if(nums.size() == 2) return max(nums[0], nums[1]);
        
        // pre：前i-2间房最大金额，cur：前i-1间房最大金额
        int pre = nums[0];
        int cur = max(nums[0], nums[1]);
        
        // 从第3间开始递推
        for(int i = 2; i < nums.size(); i++){
            // 当前最大金额：不偷当前房=cur，偷当前房=pre+nums[i]
            int next = max(cur, pre + nums[i]);
            pre = cur;
            cur = next;
        }
        return cur;
    }
};
```





## [213. 打家劫舍 II - 力扣（LeetCode）](https://leetcode.cn/problems/house-robber-ii/description/)

![屏幕截图 2026-03-25 214243.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-03-25%20214243.png?raw=true)

```cpp
class Solution {
public:
    int rob(vector<int>& nums) {
        // 只有1间直接偷
        if(nums.size() == 1) return nums[0];
        // 环形分两种情况：不偷最后一间 / 不偷第一间
        int result1 = r(nums, 0, nums.size() - 1);
        int result2 = r(nums, 1, nums.size());
        return max(result1, result2);
    }
private:
    // 区间打家劫舍：[start, end) 线性房屋
    int r(vector<int>& nums, int start, int end){
        int n = end - start;
        if(n == 1) return nums[start];
        if(n == 2) return max(nums[start], nums[start+1]);
        
        // pre：前i-2间最大值，cur：前i-1间最大值
        int pre = nums[start];
        int cur = max(nums[start], nums[start+1]);
        
        for(int i = start+2; i < end; i++){
            // 递推：不偷当前 / 偷当前
            int next = max(cur, pre + nums[i]);
            pre = cur;
            cur = next;
        }
        return cur;
    }
};
```
