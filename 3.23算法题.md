## [1594. 矩阵的最大非负积 - 力扣（LeetCode）](https://leetcode.cn/problems/maximum-non-negative-product-in-a-matrix/description/?envType=daily-question&envId=2026-03-23)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.23-1-2026-03-23-20%3A24%3A46.png)

```cpp
class Solution {
public:
    int maxProductPath(vector<vector<int>>& grid) {
       const int mod = 1000000000 + 7;
       int m = grid.size(), n = grid[0].size();
       vector<vector<long long>> maxgt(m, vector<long long>(n));//记录位置的最大值
       vector<vector<long long>> minlt(m, vector<long long>(n));//记录位置的最小值
       maxgt[0][0] = minlt[0][0] = grid[0][0];//初始化
       for(int i = 1; i < n; ++i){
        maxgt[0][i] = minlt[0][i] = maxgt[0][i - 1] * grid[0][i];//处理第一行的结果
       }
       for(int i = 1; i < m; ++i){
        maxgt[i][0] = minlt[i][0] = maxgt[i - 1][0] * grid[i][0];//处理第一列结果
       }
        for(int i = 1; i < m; ++i){
            for(int j = 1; j < n; ++j){
                if(grid[i][j] > 0){//当该位置的值为正，寻找最大值为最大
                    maxgt[i][j] = max(maxgt[i][j - 1], maxgt[i - 1][j]) * grid[i][j];
                    minlt[i][j] = min(minlt[i][j - 1], minlt[i - 1][j]) * grid[i][j];
                }
                else{//当该位置值为负，寻找最小值为最大
                    maxgt[i][j] = min(minlt[i][j - 1], minlt[i - 1][j]) * grid[i][j];
                    minlt[i][j] = max(maxgt[i][j - 1], maxgt[i - 1][j]) * grid[i][j];
                }
            }
        }
        if(maxgt[m - 1][n - 1] < 0){//若到最终仍未负，则没有最大正值
            return -1;
        }
        else{
            return maxgt[m - 1][n - 1] % mod;
        }
    }
};

```

## [2. 两数相加 - 力扣（LeetCode）](https://leetcode.cn/problems/add-two-numbers/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.23-2-2026-03-23-20%3A38%3A20.png)

```cpp
class Solution {
public:
    ListNode* addTwoNumbers(ListNode* l1, ListNode* l2) {
        ListNode dummy;//创建哑节点
        ListNode* cur = &dummy;//让cur指向哑节点
        int car = 0;//进位为0
        while(l1 || l2 || car){//当任意一个未处理完时继续处理
            if(l1){
                car += l1->val;//当前值为进位加两个链表上的数
                l1 = l1->next;//向下移动
            }
            if(l2){
                car +=l2->val;
                l2 = l2->next;
            }
            cur->next = new ListNode(car % 10);//当前位数值为取余的结果
            cur = cur->next;//cur处理下一个值
            car /= 10;//此时进位为处理后的结果
        }
        return dummy.next;
    }
};
```
