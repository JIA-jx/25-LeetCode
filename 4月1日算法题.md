## [3. 无重复字符的最长子串 - 力扣（LeetCode）](https://leetcode.cn/problems/longest-substring-without-repeating-characters/description/)

![屏幕截图 2026-04-01 185846.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-04-01%20185846.png?raw=true)

```cpp
class Solution {
public:
    int lengthOfLongestSubstring(string s) {
        int n = s.size();
        int maxLen = 0; // 最长子串长度
        int i = 0, j = -1; // 滑动窗口[i,j]
        bool hash[256] = {0}; // 字符是否在窗口内

        // 窗口右边界扩展
        while (j + 1 < n) {
            // 右字符不重复，加入窗口
            if (!hash[s[j+1]]) {
                hash[s[j + 1]] = true;
                j++; 
                maxLen = max(maxLen, j - i + 1);
            } 
            // 重复，左边界右移
            else { 
                hash[s[i]] = false;
                i++;
            }
        }
        return maxLen;
    }
};
```

## [127. 单词接龙 - 力扣（LeetCode）](https://leetcode.cn/problems/word-ladder/)

![屏幕截图 2026-04-01 190121.png](https://github.com/zhanglihang666/leetcodeImage/blob/main/%E5%B1%8F%E5%B9%95%E6%88%AA%E5%9B%BE%202026-04-01%20190121.png?raw=true)

```cpp
class Solution {
public:
    int ladderLength(string beginStr, string endStr, vector<string>& strList) {
        // 单词集合，快速查找
        unordered_set<string> strListSet(strList.begin(), strList.end());

        // 终点不存在 / 起点终点相同，直接返回0
        if(strListSet.count(endStr) == 0 || beginStr == endStr) return 0;

        unordered_set<string> visitedset; // 已访问单词
        queue<string> que;                // BFS队列
        que.push(beginStr);
        visitedset.insert(beginStr);

        int pathCount = 1; // 路径长度
        while(!que.empty()){
            int queSize = que.size();
            // 层序遍历BFS
            for(int i = 0; i < queSize; i++){
                string word = que.front();
                que.pop();

                // 逐位替换字符
                for(int j = 0; j < word.size(); j++){
                    string newWord = word;
                    for(char c = 'a'; c <= 'z'; c++){
                        newWord[j] = c;

                        // 找到终点，返回长度
                        if(newWord == endStr){
                            return pathCount + 1;
                        }

                        // 合法且未访问，加入队列
                        if(strListSet.count(newWord) && !visitedset.count(newWord)){
                            visitedset.insert(newWord);
                            que.push(newWord);
                        }
                    }
                }
            }
            pathCount++; // 层数+1
        }
        return 0;
    }
};
```
