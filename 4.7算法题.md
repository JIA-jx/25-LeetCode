## [35. 搜索插入位置 - 力扣（LeetCode）](https://leetcode.cn/problems/search-insert-position/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/4.7-1-2026-04-07-21%3A06%3A44.png)

```cpp
class Solution {
public:
    int searchInsert(vector<int>& nums, int target) {
        int left = 0;//左指针放前面
        int right = nums.size() - 1;//右指针放后面
        if(target < nums[0]){//插入最前面
            return 0;
        }
        else if(target > nums[right]){//插入最后面
            return right + 1;
        }
        else{
            while(left <= right){//二分查找
                int middle = left +((right - left) / 2);
                if(nums[middle] > target){
                    right = middle - 1;
                }
                else if(nums[middle] < target){
                    left = middle + 1;
                }
                else{
                    return middle;
                }
            }
            return left;
        }
    }
};
```

## [153. 寻找旋转排序数组中的最小值 - 力扣（LeetCode）](https://leetcode.cn/problems/find-minimum-in-rotated-sorted-array/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/4.7-2-2026-04-07-21%3A10%3A23.png)

```cpp
class Solution {
public:
    int findMin(vector<int>& nums) {
        int left = 0;
        int right = nums.size() - 1;
        while(left < right){//二分查找
            int mid = left + (right - left) / 2;//取中间值
            if(nums[mid] <nums[right]){//忽略右半
                right = mid;
            }
            else{//忽略左半
                left = mid +1;
            }
        }
        return nums[left];
    }
};
```
