## [41. 缺失的第一个正数 - 力扣（LeetCode）](https://leetcode.cn/problems/first-missing-positive/submissions/708345389/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.17-1-2026-03-17-16%3A43%3A26.png)

```cpp
class Solution {
public:
    int firstMissingPositive(vector<int>& nums) {
        int n = nums.size();
        for (int i = 0; i < n; ++i) {
            //处理大于零，小于n的数据
            while (nums[i] > 0 && nums[i] <= n && nums[nums[i] - 1] != nums[i]) {
                swap(nums[i], nums[nums[i] - 1]);//索引+1 =值 
            }
        }
        for (int i = 0; i < n; ++i) {
            if (nums[i] != i + 1) {//索引与值不符
                return i + 1;//返回应得的值
            }
        }
        return n + 1;//否则为数组下一个数
    }
};
```

## [73. 矩阵置零 - 力扣（LeetCode）](https://leetcode.cn/problems/set-matrix-zeroes/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.17-2-2026-03-17-17%3A03%3A53.png)

```cpp
class Solution {
public:
    void setZeroes(vector<vector<int>>& matrix) {
        int n = matrix[0].size(),m = matrix.size();
        vector<int> row(m,0),col(n,0);//建立标记行、列的数组
        for(int i = 0; i < m; ++i){
            for(int j = 0; j < n; ++j){
                if(matrix[i][j] == 0){
                    row[i] = col[j] = 1;//标记需要清0的行和列
                }
            }
        }
        for(int i = 0; i < m;++i){
            for(int j = 0; j < n; ++j){
                if(row[i] || col[j]){//清0
                    matrix[i][j] = 0;
                }
            }
        }
    }
};
```
