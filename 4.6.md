```c
int* path;
int pathTop;
int** res;
int resTop;
void backtracking(int n, int k, int startIndex) {
    if(pathTop == k) {
        int* temp = (int*)malloc(sizeof(int) * k);
        for(int j = 0; j < k; j++) {
            temp[j] = path[j];
        }
        res[resTop++] = temp;
        return;
    }
    for(int i = startIndex; i <= n - (k - pathTop) + 1; i++) {
        path[pathTop++] = i;
        backtracking(n, k, i + 1);
        pathTop--;
    }
}

int** combine(int n, int k, int* returnSize, int** returnColumnSizes) {
    path = (int*)malloc(sizeof(int) * k);
    res = (int**)malloc(sizeof(int*) * 1000000);

    pathTop = resTop = 0;
    backtracking(n, k, 1);

    *returnSize = resTop;

    *returnColumnSizes = (int*)malloc(sizeof(int) * (*returnSize));
    for(int i = 0; i < *returnSize; i++) {
        (*returnColumnSizes)[i] = k;
    }
    return res;
}
```

## [216. 组合总和 III ](https://leetcode.cn/problems/combination-sum-iii/submissions/715742390/)

```c
int* path;
int pathTop;
int** ans;
int ansTop;
void backtracking(int n, int k, int sum, int startIndex) {
    if(sum > n) {
        return;
    }

    if(pathTop == k) {
        if(sum == n) {
            int* temp = (int*)malloc(sizeof(int) * k);
            for(int j = 0; j < k; j++) {
                temp[j] = path[j];
            }
            ans[ansTop++] = temp;
        }
        return;
    }

    for(int i = startIndex; i <= 9 - (k - pathTop) + 1; i++) {
        sum += i;
        path[pathTop++] = i;
        backtracking(n, k, sum, i + 1);
        sum -= i;
        pathTop--;
    }
}

int** combinationSum3(int k, int n, int* returnSize, int** returnColumnSizes) {
    path = (int*)malloc(sizeof(int) * k);
    ans = (int**)malloc(sizeof(int*) * 100);

    pathTop = ansTop = 0;
    backtracking(n, k, 0, 1);

    *returnSize = ansTop;
    *returnColumnSizes = (int*)malloc(sizeof(int) * ansTop);
    for(int i = 0; i < ansTop; i++) {
        (*returnColumnSizes)[i] = k;
    }

    return ans;
}
```
