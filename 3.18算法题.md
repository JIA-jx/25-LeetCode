## [3070. 元素和小于等于 k 的子矩阵的数目 - 力扣（LeetCode）](https://leetcode.cn/problems/count-submatrices-with-top-left-element-and-sum-less-than-k/description/?envType=daily-question&envId=2026-03-18)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.18-1-2026-03-18-18%3A50%3A06.png)

```cpp
class Solution {
public:
    int countSubmatrices(vector<vector<int>>& grid, int k) {
        int res = 0;//当前结果为0
        int n = grid.size();//统计行数
        int m = grid[0].size();//统计列数
        vector<int> cols(m);//记录列的前缀和
        for(int i = 0; i < n; ++i){
            int rows = 0;//清空上次统计结果
            for(int j = 0; j < m; ++j){
                cols[j] += grid[i][j];//统计第i行时j的前缀和
                rows += cols[j];//从0行0列到第i行j列的值
                if( rows <=k){//符合条件
                    res++;
                }
            }
        } 
        return res;
    }
};
```

## [54. 螺旋矩阵 - 力扣（LeetCode）](https://leetcode.cn/problems/spiral-matrix/description/?envType=study-plan-v2&envId=top-100-liked)

![](https://raw.githubusercontent.com/ningmeng2134/git-demo/main/3.18-2-2026-03-18-19%3A38%3A10.png)

```cpp
class Solution {
public:
    vector<int> spiralOrder(vector<vector<int>>& matrix) {
        int dy[4] = {1,0,-1,0};//y的方向数组
        int dx[4] = {0,1,0,-1};//x的方向数组
        if(matrix.empty() || matrix[0].empty()) return{};//处理空数组
        int m = matrix[0].size();
        int n = matrix.size();
        vector<vector<bool>> t(n,vector<bool>(m,false));//创建布尔数组标记访问记录
        vector<int> res;//结果数组
        int x = 0, y = 0;
        int d = 0;//方向
        for(int i = 0; i < m*n; ++i){
            res.push_back(matrix[x][y]);
            t[x][y] = true;//标记被访问
            int tx = x + dx[d];//尝试进入下一个数值
            int ty = y + dy[d];
            if(tx >= n || tx < 0 || ty < 0 || ty >= m || t[tx][ty] == true){//超过边界或被访问过
                d = (d + 1) % 4;//改变方向进行遍历
                tx = x + dx[d];//调整至正确方向
                ty = y + dy[d];
            }
            x =tx;//让x、y成为正确的索引
            y = ty;
        }
        return res;
    }
};
```
