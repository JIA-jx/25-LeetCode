## [17. 电话号码的字母组合](https://leetcode.cn/problems/letter-combinations-of-a-phone-number/submissions/706107249/)

![image-20260312153720475](C:\Users\OMEN\AppData\Roaming\Typora\typora-user-images\image-20260312153720475.png)

```c
char letterMap[10][5] = {"\0", "\0", "abc", "def", "ghi", "jkl", "mno", "pqrs", "tuv", "wxyz"};
char* path;
int pathTop;
char** res;
int resTop;

void backtracking(char* digits, int index) {
    //若当前下标等于digits长度
    if(index == strlen(digits)) {
        char* temp = (char*)malloc(sizeof(char) * (index + 1));
        for(int j = 0; j < pathTop; j++) {
            temp[j] = path[j];
        }
        temp[index] = '\0';   //在字符串末尾添加'\0'
        res[resTop++] = temp;  //存入结果
        return;
    }

    int digit = digits[index] - '0';   //将字符型转化为整型
    char* letters = letterMap[digit];

    for(int i = 0; i < strlen(letters); i++) {
        path[pathTop++] = letters[i]; 
        backtracking(digits, index + 1);
        pathTop--;   //回溯
    }
}
 
char** letterCombinations(char* digits, int* returnSize) {
    path = (char*)malloc(sizeof(char) * strlen(digits));
    res = (char**)malloc(sizeof(char*) * 200);

    pathTop = resTop = 0;
    backtracking(digits, 0);

    *returnSize = resTop;
    return res;
}
```



## [39. 组合总和](https://leetcode.cn/problems/combination-sum/submissions/706170113/)

![image-20260312165757388](3.12.assets/image-20260312165757388.png)

```c
int* path;
int pathTop;
int** res;
int resTop;
int sum;
int* length;  //记录各数组的长度

void backtracking(int* candidates, int candidatesSize, int target, int sum, int index) {
    if(sum > target) {  //如果sum已经大于target，直接返回
        return;
    }

    if(sum == target) {
        int* temp = (int*)malloc(sizeof(int) * pathTop);
        for(int j = 0; j < pathTop; j++) {
            temp[j] = path[j];
        }
        length[resTop] = pathTop;
        res[resTop++] = temp;
        
        return;
    }

    for(int i = index; i < candidatesSize; i++) {  //从index索引开始，防止取到之前出现过的结果
        path[pathTop++] = candidates[i];
        sum += candidates[i];
        backtracking(candidates, candidatesSize, target, sum, i);
        //回溯
        pathTop--;
        sum -= candidates[i];
    }
}

int** combinationSum(int* candidates, int candidatesSize, int target, int* returnSize, int** returnColumnSizes) {
    path = (int*)malloc(sizeof(int) * 40);
    res = (int**)malloc(sizeof(int*) * 200);
    length = (int*)malloc(sizeof(int) * 200);

    pathTop = resTop = 0;
    backtracking(candidates, candidatesSize, target, 0, 0);

    *returnSize = resTop;
    *returnColumnSizes = (int*)malloc(sizeof(int) * (*returnSize));
    for(int i = 0; i < resTop; i++) {
        (*returnColumnSizes)[i] = length[i];
    }

    return res;
}
```
